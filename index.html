// مصفوفة عشان نحفظ فيها الطلبات let cart = []; // متغير عشان نجمع فيه الحساب الكلي let totalAmount = 0; // دالة إضافة الأصناف العادية (اللي ليها سعر واحد) function addToCart(itemName, price) { // إضافة الصنف للمصفوفة cart.push({ name: itemName, price: price }); // تزويد السعر على الإجمالي totalAmount += price; // تحديث شكل الفاتورة قدام الزبون updateCartUI(); } // دالة إضافة الأصناف اللي ليها حجمين (الصغير والكبير) function addBoxToCart(itemName, selectElementId) { // بنجيب القائمة اللي الزبون اختار منها let select = document.getElementById(selectElementId); let option = select.options[select.selectedIndex]; // بنسحب السعر واسم الحجم let price = Number(option.value); let sizeName = option.getAttribute("data-name"); // بنجمع اسم الصنف مع الحجم (مثال: مكس فول وباذنجان (صغير)) let fullName = itemName + " (" + sizeName + ")"; // بنبعت الصنف لدالة الإضافة الأساسية addToCart(fullName, price); } // دالة تحديث سلة المشتريات عشان تظهر للزبون function updateCartUI() { let cartList = document.getElementById("cart-items"); let totalDisplay = document.getElementById("total-amount"); // تفريغ القائمة القديمة عشان نرسمها من جديد cartList.innerHTML = ""; // لفة على كل الأصناف اللي الزبون طلبها عشان نعرضها cart.forEach(function(item) { let listItem = document.createElement("li"); let nameSpan = document.createElement("span"); nameSpan.textContent = item.name; let priceSpan = document.createElement("span"); priceSpan.textContent = item.price + " ج"; priceSpan.style.fontWeight = "bold"; listItem.appendChild(nameSpan); listItem.appendChild(priceSpan); cartList.appendChild(listItem); }); // تحديث رقم الإجمالي تحت totalDisplay.textContent = totalAmount; }
